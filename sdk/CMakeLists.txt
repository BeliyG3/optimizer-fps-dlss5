add_library(ofps_sdk_core STATIC
    src/input_v2.cpp src/layout.cpp src/layout_v2.cpp
    src/layout_v2_sides.cpp src/layout_v2_validation.cpp src/math.cpp)
add_library(OptimizerFps::SdkCore ALIAS ofps_sdk_core)
set_target_properties(ofps_sdk_core PROPERTIES EXPORT_NAME SdkCore)
target_compile_features(ofps_sdk_core PUBLIC cxx_std_20)
target_include_directories(ofps_sdk_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
target_compile_options(ofps_sdk_core PRIVATE /W4 /permissive-)
else()
target_compile_options(ofps_sdk_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(OFPS_BUILD_D3D_ADAPTERS)
if(NOT WIN32)
message(FATAL_ERROR "Direct3D adapters are available only on Windows")
endif()
add_subdirectory(adapters/d3d11)
add_subdirectory(adapters/d3d12)
endif()
set(_ofps_sdk_install_targets ofps_sdk_core)
if(TARGET ofps_sdk_d3d11)
list(APPEND _ofps_sdk_install_targets ofps_sdk_d3d11)
endif()
if(TARGET ofps_sdk_d3d12)
list(APPEND _ofps_sdk_install_targets ofps_sdk_d3d12)
endif()
install(TARGETS ${_ofps_sdk_install_targets} EXPORT OptimizerFpsSdkTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(OFPS_BUILD_D3D_ADAPTERS)
install(FILES adapters/common/adapter_types.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/optimizer_fps/adapters/common)
install(FILES adapters/d3d11/d3d11_adapter.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/optimizer_fps/adapters/d3d11)
install(FILES adapters/d3d12/d3d12_adapter.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/optimizer_fps/adapters/d3d12)
endif()
install(DIRECTORY shaders/ DESTINATION ${CMAKE_INSTALL_DATADIR}/optimizer-fps-sdk/shaders FILES_MATCHING PATTERN "*.hlsl" PATTERN "*.hlsli")
